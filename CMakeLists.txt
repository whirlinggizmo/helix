cmake_minimum_required(VERSION 3.16)
project(helix_asteroids LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "This first native adapter's hxcpp linking and verification are Linux-only")
endif()
find_package(X11 REQUIRED)
find_package(ALSA REQUIRED)
find_library(SNDFILE_LIBRARY NAMES sndfile libsndfile.so.1 REQUIRED)
file(WRITE "${CMAKE_BINARY_DIR}/codec-link.json" "{\"library\":\"${SNDFILE_LIBRARY}\"}\n")
set(HY_SERVICES src/helix/adapters/input/sokol/internal/input.c src/helix/adapters/input/evdev/internal/gamepad_linux.c src/helix/adapters/audio/sokol/internal/audio.c)
add_library(helix STATIC src/helix/adapters/host/sokol/internal/host.c src/helix/adapters/graphics/sokol/internal/graphics.c src/helix/adapters/audio/libsndfile/internal/codec.c ${HY_SERVICES})
target_compile_features(helix PRIVATE c_std_11)
target_include_directories(helix PRIVATE src/helix/adapters/graphics/sokol/internal src/helix/adapters/host/sokol/internal src/helix/adapters/audio/libsndfile/internal src/helix/adapters/audio/sokol/internal src/helix/adapters/input/sokol/internal src/helix/adapters/input/evdev/internal vendor build/generated/asteroids ${X11_INCLUDE_DIR} ${ALSA_INCLUDE_DIRS})
target_compile_definitions(helix PRIVATE _POSIX_C_SOURCE=200809L)
set_target_properties(helix PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_executable(native-services-test tests/native/native_services.c ${HY_SERVICES})
target_compile_features(native-services-test PRIVATE c_std_11)
target_compile_options(native-services-test PRIVATE -UNDEBUG)
target_compile_definitions(native-services-test PRIVATE _POSIX_C_SOURCE=200809L HY_SERVICES_TEST SOKOL_DUMMY_BACKEND)
target_include_directories(native-services-test PRIVATE src/helix/adapters/graphics/sokol/internal src/helix/adapters/host/sokol/internal src/helix/adapters/audio/libsndfile/internal src/helix/adapters/audio/sokol/internal src/helix/adapters/input/sokol/internal src/helix/adapters/input/evdev/internal vendor)
target_link_libraries(native-services-test PRIVATE m pthread)
add_executable(native-audio-probe tests/native/native_audio_probe.c src/helix/adapters/audio/sokol/internal/audio.c)
target_compile_features(native-audio-probe PRIVATE c_std_11)
target_compile_definitions(native-audio-probe PRIVATE _POSIX_C_SOURCE=200809L)
target_include_directories(native-audio-probe PRIVATE src/helix/adapters/graphics/sokol/internal src/helix/adapters/host/sokol/internal src/helix/adapters/audio/libsndfile/internal src/helix/adapters/audio/sokol/internal src/helix/adapters/input/sokol/internal src/helix/adapters/input/evdev/internal vendor ${ALSA_INCLUDE_DIRS})
target_link_libraries(native-audio-probe PRIVATE ${ALSA_LIBRARIES} m pthread)

# Test-only GL call interposer; excluded from native/browser delivery and all timing.
add_library(native-graphics-call-audit SHARED tests/native/graphics_calls.c)
target_compile_features(native-graphics-call-audit PRIVATE c_std_11)
target_link_libraries(native-graphics-call-audit PRIVATE dl)
