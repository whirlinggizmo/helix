cmake_minimum_required(VERSION 3.16)
project(hybrid_asteroids LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "This first native adapter's hxcpp linking and verification are Linux-only")
endif()
find_package(X11 REQUIRED)
find_package(ALSA REQUIRED)
find_library(SNDFILE_LIBRARY NAMES sndfile libsndfile.so.1 REQUIRED)
file(WRITE "${CMAKE_BINARY_DIR}/codec-link.json" "{\"library\":\"${SNDFILE_LIBRARY}\"}\n")
set(HY_SERVICES native/services.c native/gamepad_linux.c native/audio.c)
add_library(hybrid STATIC native/hybrid.c native/codec.c ${HY_SERVICES})
target_compile_features(hybrid PRIVATE c_std_11)
target_include_directories(hybrid PRIVATE native vendor build/generated ../../02-haxe-sokol-hal/prototype/vendor ${X11_INCLUDE_DIR} ${ALSA_INCLUDE_DIRS})
target_compile_definitions(hybrid PRIVATE _POSIX_C_SOURCE=200809L)
set_target_properties(hybrid PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_executable(native-services-test tests/native_services.c ${HY_SERVICES})
target_compile_features(native-services-test PRIVATE c_std_11)
target_compile_options(native-services-test PRIVATE -UNDEBUG)
target_compile_definitions(native-services-test PRIVATE _POSIX_C_SOURCE=200809L HY_SERVICES_TEST SOKOL_DUMMY_BACKEND)
target_include_directories(native-services-test PRIVATE native vendor ../../02-haxe-sokol-hal/prototype/vendor)
target_link_libraries(native-services-test PRIVATE m pthread)
add_executable(native-audio-probe tests/native_audio_probe.c native/audio.c)
target_compile_features(native-audio-probe PRIVATE c_std_11)
target_compile_definitions(native-audio-probe PRIVATE _POSIX_C_SOURCE=200809L)
target_include_directories(native-audio-probe PRIVATE native vendor ../../02-haxe-sokol-hal/prototype/vendor ${ALSA_INCLUDE_DIRS})
target_link_libraries(native-audio-probe PRIVATE ${ALSA_LIBRARIES} m pthread)
